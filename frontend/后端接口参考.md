## 基础实现

```python
from fastapi import FastAPI, File, UploadFile, Form, HTTPException
from fastapi.responses import FileResponse
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from typing import Optional
import uuid
import os
import zipfile
import shutil
from datetime import datetime
import asyncio

app = FastAPI(title="3DGS API")

# CORS 配置
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 配置
UPLOAD_DIR = "uploads"
MODEL_DIR = "models"
OPTIMIZED_DIR = "optimized"

os.makedirs(UPLOAD_DIR, exist_ok=True)
os.makedirs(MODEL_DIR, exist_ok=True)
os.makedirs(OPTIMIZED_DIR, exist_ok=True)

# 存储任务状态
tasks = {}
models = {}

class OptimizeRequest(BaseModel):
    modelId: str
    algorithm: str
    compressionLevel: str
    qualityFirst: bool

def validate_zip_file(zip_path: str) -> bool:
    """验证 ZIP 文件合法性"""
    try:
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            files = zip_ref.namelist()
            
            # 过滤出图像文件
            image_extensions = ('.jpg', '.jpeg', '.png', '.JPG', '.JPEG', '.PNG')
            image_files = [f for f in files if f.lower().endswith(image_extensions)]
            
            # 至少需要 20 张图像
            if len(image_files) < 20:
                return False
            
            # 检查是否有隐藏文件或非法路径
            for f in files:
                if '..' in f or f.startswith('/'):
                    return False
            
            return True
    except:
        return False

@app.post("/api/3dgs/generate")
async def generate_3dgs(
    file: UploadFile = File(...),
    modelName: str = Form(...),
    iterations: int = Form(7000),
    resolution: int = Form(1)
):
    """生成 3DGS 模型"""
    
    # 验证文件类型
    if not file.filename.endswith('.zip'):
        raise HTTPException(status_code=400, detail="只支持 ZIP 文件")
    
    # 保存上传的文件
    task_id = str(uuid.uuid4())
    upload_path = os.path.join(UPLOAD_DIR, f"{task_id}.zip")
    
    with open(upload_path, "wb") as f:
        content = await file.read()
        f.write(content)
    
    # 验证 ZIP 文件
    if not validate_zip_file(upload_path):
        os.remove(upload_path)
        raise HTTPException(status_code=400, detail="ZIP 文件不合法或图像数量不足")
    
    # 创建任务
    tasks[task_id] = {
        "taskId": task_id,
        "status": "processing",
        "progress": 0,
        "log": "正在解压文件...\n",
        "modelName": modelName,
        "iterations": iterations,
        "resolution": resolution
    }
    
    # 异步处理生成任务
    asyncio.create_task(process_generation(task_id, upload_path, modelName, iterations, resolution))
    
    return {"code": 200, "data": {"taskId": task_id}}

async def process_generation(task_id: str, zip_path: str, model_name: str, iterations: int, resolution: int):
    """处理 3DGS 生成（后台任务）"""
    try:
        # 解压文件
        extract_dir = os.path.join(UPLOAD_DIR, task_id)
        os.makedirs(extract_dir, exist_ok=True)
        
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extractall(extract_dir)
        
        tasks[task_id]["progress"] = 10
        tasks[task_id]["log"] += "文件解压完成\n正在运行 COLMAP...\n"
        
        # 这里调用实际的 3DGS 训练流程
        # 示例命令（需要根据实际环境调整）:
        # 1. COLMAP 特征提取和匹配
        # 2. 3DGS 训练
        # 类似于：
        # subprocess.run([
        #     "python", "train.py",
        #     "-s", extract_dir,
        #     "-m", output_dir,
        #     "--iterations", str(iterations),
        #     "--resolution", str(resolution)
        # ])
        
        # 模拟处理过程
        for i in range(10, 100, 10):
            await asyncio.sleep(2)
            tasks[task_id]["progress"] = i
            tasks[task_id]["log"] += f"训练进度: {i}%\n"
        
        # 保存模型信息
        model_id = str(uuid.uuid4())
        model_path = os.path.join(MODEL_DIR, f"{model_id}.ply")
        
        models[model_id] = {
            "id": model_id,
            "name": model_name,
            "size": os.path.getsize(zip_path),
            "status": "completed",
            "createdAt": datetime.now().isoformat(),
            "iterations": iterations,
            "resolution": resolution,
            "pointCount": 100000,  # 示例值
            "path": model_path
        }
        
        tasks[task_id]["status"] = "completed"
        tasks[task_id]["progress"] = 100
        tasks[task_id]["log"] += "生成完成！\n"
        tasks[task_id]["modelId"] = model_id
        
    except Exception as e:
        tasks[task_id]["status"] = "failed"
        tasks[task_id]["log"] += f"错误: {str(e)}\n"

@app.get("/api/3dgs/status/{task_id}")
async def get_generate_status(task_id: str):
    """获取生成任务状态"""
    if task_id not in tasks:
        raise HTTPException(status_code=404, detail="任务不存在")
    
    return {"code": 200, "data": tasks[task_id]}

@app.get("/api/3dgs/models")
async def get_models():
    """获取模型列表"""
    return {"code": 200, "data": list(models.values())}

@app.get("/api/3dgs/model/{model_id}")
async def get_model_info(model_id: str):
    """获取模型详情"""
    if model_id not in models:
        raise HTTPException(status_code=404, detail="模型不存在")
    
    return {"code": 200, "data": models[model_id]}

@app.delete("/api/3dgs/model/{model_id}")
async def delete_model(model_id: str):
    """删除模型"""
    if model_id not in models:
        raise HTTPException(status_code=404, detail="模型不存在")
    
    model = models[model_id]
    if os.path.exists(model["path"]):
        os.remove(model["path"])
    
    del models[model_id]
    return {"code": 200}

@app.post("/api/3dgs/optimize")
async def optimize_model(request: OptimizeRequest):
    """优化模型"""
    if request.modelId not in models:
        raise HTTPException(status_code=404, detail="模型不存在")
    
    task_id = str(uuid.uuid4())
    tasks[task_id] = {
        "taskId": task_id,
        "status": "processing",
        "progress": 0,
        "log": "正在加载模型...\n",
        "algorithm": request.algorithm
    }
    
    asyncio.create_task(process_optimization(task_id, request))
    
    return {"code": 200, "data": {"taskId": task_id}}

async def process_optimization(task_id: str, request: OptimizeRequest):
    """处理优化任务"""
    try:
        model = models[request.modelId]
        tasks[task_id]["log"] += f"使用 {request.algorithm} 算法进行优化...\n"
        
        # 这里调用实际的压缩算法
        # 根据 algorithm 选择不同的实现:
        # - lightgaussian: LightGaussian 压缩
        # - mini-splatting: Mini-Splatting 压缩
        # - compact3d: Compact3D 压缩
        
        # 模拟处理
        for i in range(10, 100, 10):
            await asyncio.sleep(1)
            tasks[task_id]["progress"] = i
            tasks[task_id]["log"] += f"优化进度: {i}%\n"
        
        # 计算压缩结果
        original_size = model["size"]
        compression_map = {"low": 0.6, "medium": 0.4, "high": 0.2}
        ratio = compression_map.get(request.compressionLevel, 0.4)
        optimized_size = int(original_size * ratio)
        
        tasks[task_id]["status"] = "completed"
        tasks[task_id]["progress"] = 100
        tasks[task_id]["log"] += "优化完成！\n"
        tasks[task_id]["comparison"] = {
            "originalSize": original_size,
            "optimizedSize": optimized_size,
            "compressionRatio": round((1 - ratio) * 100, 2),
            "psnr": 32.5  # 示例质量分数
        }
        
    except Exception as e:
        tasks[task_id]["status"] = "failed"
        tasks[task_id]["log"] += f"错误: {str(e)}\n"

@app.get("/api/3dgs/optimize/status/{task_id}")
async def get_optimize_status(task_id: str):
    """获取优化任务状态"""
    if task_id not in tasks:
        raise HTTPException(status_code=404, detail="任务不存在")
    
    return {"code": 200, "data": tasks[task_id]}

@app.get("/api/3dgs/file/{model_id}")
async def get_model_file(model_id: str):
    """获取模型文件"""
    if model_id not in models:
        raise HTTPException(status_code=404, detail="模型不存在")
    
    model = models[model_id]
    if not os.path.exists(model["path"]):
        raise HTTPException(status_code=404, detail="模型文件不存在")
    
    return FileResponse(model["path"])

@app.get("/api/3dgs/download/{task_id}")
async def download_optimized(task_id: str):
    """下载优化后的模型"""
    if task_id not in tasks:
        raise HTTPException(status_code=404, detail="任务不存在")
    
    # 返回优化后的文件
    # 实际实现需要根据任务找到对应的优化文件
    optimized_path = os.path.join(OPTIMIZED_DIR, f"{task_id}.ply")
    
    if not os.path.exists(optimized_path):
        raise HTTPException(status_code=404, detail="优化文件不存在")
    
    return FileResponse(optimized_path, filename=f"optimized_{task_id}.ply")

if __name__ == "__main__":
```

## 集成实际的 3DGS 训练

需要集成实际的 3DGS 训练代码，推荐使用：

2. **Docker 容器**: 可以将 3DGS 训练打包成 Docker 服务

## 集成压缩算法

根据选择的算法，需要：

1. Clone 对应的 GitHub 仓库
2. 配置 Python 环境
3. 在 `process_optimization` 函数中调用对应的脚本

示例：

```python
import subprocess

# LightGaussian
subprocess.run([
    "python", "path/to/LightGaussian/compress.py",
    "--model_path", model_path,
    "--output_path", output_path,
    "--compression_level", compression_level
])
```

## 注意事项

1. 实际部署需要配置 GPU 环境
2. 需要足够的磁盘空间存储模型
3. 建议使用任务队列（如 Celery）处理长时间运行的任务
4. 添加用户认证和权限管理
5. 实现文件清理机制，定期删除临时文件
