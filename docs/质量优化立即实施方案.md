def clean_gaussians(ply_path, opacity_threshold=0.1, distance_percentile=95):
    """清理高斯点：删除低不透明度和离群点"""
    from plyfile import PlyData
    import numpy as np
    
    # 读取 PLY
    plydata = PlyData.read(ply_path)
    vertex = plydata['vertex']
    
    # 提取数据
    xyz = np.stack([vertex['x'], vertex['y'], vertex['z']], axis=1)
    opacity = vertex['opacity']
    
    # 应用 sigmoid
    opacity = 1 / (1 + np.exp(-opacity))
    
    # 过滤 1: 不透明度
    mask_opacity = opacity >= opacity_threshold
    
    # 过滤 2: 距离中心
    center = np.median(xyz, axis=0)
    distances = np.linalg.norm(xyz - center, axis=1)
    dist_threshold = np.percentile(distances, distance_percentile)
    mask_distance = distances <= dist_threshold
    
    # 组合过滤
    mask = mask_opacity & mask_distance
    
    print(f"清理前: {len(xyz)} 个高斯")
    print(f"删除: {(~mask).sum()} 个 ({(~mask).sum() / len(xyz) * 100:.1f}%)")
    print(f"保留: {mask.sum()} 个")
    
    # 保存清理后的模型
    cleaned_vertex = vertex[mask]
    from plyfile import PlyElement
    cleaned_plydata = PlyData([PlyElement.describe(cleaned_vertex, 'vertex')])
    cleaned_plydata.write(ply_path)
    
    return mask.sum(), (~mask).sum()
```

def apply_adaptive_params(params_dict, user_params):
    """
    应用自适应参数，用户参数优先
    
    参数:
    - params_dict: 推荐的参数字典
    - user_params: 用户提供的参数（可能为 None）
    
    返回: 合并后的参数
    """
    result = params_dict.copy()
    
    # 用户参数覆盖推荐参数
    if user_params:
        for key, value in user_params.items():
            if value is not None:
                result[key] = value
    
    return result

"""质量评分系统"""

def calculate_quality_score(stats, params):
    """
    计算质量评分（0-100）
    
    参数:
    - stats: train_stats.json 的内容
    - params: 训练参数
    
    返回:
    {
        "score": 92,
        "rating": "优秀",
        "factors": {
            "loss": {"score": 95, "value": 0.0155, "weight": 0.4},
            "gaussians": {"score": 90, "value": 261123, "weight": 0.3},
            ...
        },
        "recommendations": ["质量已经很好", "如需更高清晰度，可尝试 downsample=1"]
    }
    """
    score = 100
    factors = {}
    recommendations = []
    
    # 因子 1: Loss（权重 40%）
    loss = stats.get("final_loss", 1.0)
    loss_score = 100
    if loss > 0.10:
        loss_score = 0
        recommendations.append("Loss 过高，建议增加训练迭代次数或改善拍摄质量")
    elif loss > 0.05:
        loss_score = 50
        recommendations.append("Loss 较高，可增加 iterations 到 5000-7000")
    elif loss > 0.02:
        loss_score = 80
    elif loss > 0.01:
        loss_score = 95
    
    factors["loss"] = {
        "score": loss_score,
        "value": loss,
        "weight": 0.4,
        "description": "训练收敛程度"
    }
    score -= (100 - loss_score) * 0.4
    
    # 因子 2: 高斯数量（权重 30%）
    n_gaussians = stats.get("num_gaussians", 0)
    gaussians_score = 100
    if n_gaussians < 50000:
        gaussians_score = 20
        recommendations.append("高斯点过少，可能重建失败，检查图片质量和数量")
    elif n_gaussians < 100000:
        gaussians_score = 60
    elif n_gaussians > 1500000:
        gaussians_score = 70
        recommendations.append("高斯点过多，建议减小 cap_max 或启用自动清理")
    elif n_gaussians > 1200000:
        gaussians_score = 85
    
    factors["gaussians"] = {
        "score": gaussians_score,
        "value": n_gaussians,
        "weight": 0.3,
        "description": "高斯点数量合理性"
    }
    score -= (100 - gaussians_score) * 0.3
    
    # 因子 3: 显存效率（权重 20%）
    gpu_mem = stats.get("peak_gpu_mem_gb", 0)
    mem_score = 100
    if gpu_mem > 7.5:
        mem_score = 60
        recommendations.append("显存占用接近上限，考虑增大 downsample 或减小 cap_max")
    elif gpu_mem > 7.0:
        mem_score = 80
    
    factors["memory"] = {
        "score": mem_score,
        "value": gpu_mem,
        "weight": 0.2,
        "description": "显存使用效率"
    }
    score -= (100 - mem_score) * 0.2
    
    # 因子 4: 训练效率（权重 10%）
    train_time = stats.get("train_seconds", 0)
    iterations = stats.get("iterations", 1)
    speed = iterations / max(train_time, 1)  # iter/s
    
    speed_score = 100
    if speed < 10:
        speed_score = 70
    elif speed < 20:
        speed_score = 85
    
    factors["efficiency"] = {
        "score": speed_score,
        "value": round(speed, 2),
        "weight": 0.1,
        "description": "训练速度 (iter/s)"
    }
    score -= (100 - speed_score) * 0.1
    
    # 综合评分
    score = max(0, min(100, round(score)))
    
    # 评级
    if score >= 90:
        rating = "优秀"
    elif score >= 80:
        rating = "良好"
    elif score >= 70:
        rating = "一般"
    elif score >= 60:
        rating = "及格"
    else:
        rating = "较差"
    
    # 无建议时的默认提示
    if not recommendations:
        if score >= 90:
            recommendations.append("质量已经很好，无需调整")
        if params.get("downsample", 4) > 1:
            recommendations.append("如需更高清晰度，可尝试 downsample=1")
    
    return {
        "score": score,
        "rating": rating,
        "factors": factors,
        "recommendations": recommendations
    }
```
